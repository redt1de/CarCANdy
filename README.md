# CarCANdy


# SN65HVD233DR Mode select pin (Rs): 
    high speed mode:    JP 2->1 and solder RS
    slope control mode: JP 2->1 and open RS
    low power mode:     JP 2->3


# Aux Ports
    AUX1: digital, UART,PWM
        28: RX,D28
        29: TX,D28

    AUX2: digital,analog,UART,PWM
        24: TX,D24,A10
        25: RX,D25,A11
